// Synthetic GitHub: one selected private repository with per-path contents.
internal sealed class FakeGitHub : HttpMessageHandler
{
    internal const string Token = "ghu_syntheticTESTtoken00000000000000000000";
    internal long RepoId = 987, Owner = 42;
    internal string Name = "nose-calibration-submissions", Visibility = "private", OwnerType = "User";
    internal bool Public, Archived, Redirect, Paginate, OrganizerAccepted = true, OrganizerReadOnly, FailUploadAfterStore, FailUploadBeforeStore, MissingRepo;
    internal long? ExtraCollaborator, OrganizerSpoofId;
    internal HttpStatusCode? PutStatus;
    internal int Puts, Requests;
    internal string? BlockPath;
    internal Action<string>? BeforeRequest;
    internal readonly TaskCompletionSource Blocked = new(TaskCreationOptions.RunContinuationsAsynchronously);
    internal readonly Dictionary<string, byte[]> Files = new();
    internal readonly List<string> Paths = new(), Bodies = new(), Events = new();
    internal readonly List<string?> Auth = new();
    static HttpResponseMessage Json(object data, HttpStatusCode status = HttpStatusCode.OK) => new(status) { Content = new StringContent(JsonSerializer.Serialize(data)) };
    static string BlobSha(byte[] payload) => Convert.ToHexString(SHA1.HashData(Encoding.ASCII.GetBytes("blob " + payload.Length + "\0").Concat(payload).ToArray())).ToLowerInvariant();
    object RepoJson() => new { id = RepoId, name = Name, @private = !Public, visibility = Public ? "public" : Visibility, archived = Archived, disabled = false, owner = new { id = Owner, login = "participant", type = OwnerType } };
    object Person(long id, string login, bool push) => new { id, login, type = "User", permissions = new { admin = id == Owner, maintain = false, push, triage = push, pull = true } };
    protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage r, CancellationToken ct)
    {
        string p = r.RequestUri!.AbsolutePath; Requests++; Paths.Add(r.Method + " " + r.RequestUri.PathAndQuery); Events.Add(r.Method + " " + p); Auth.Add(r.Headers.Authorization?.Parameter);
        string body = r.Content == null ? "" : await r.Content.ReadAsStringAsync(ct); Bodies.Add(body);
        if (r.RequestUri.Host != "api.github.com") throw new Exception("Unexpected host");
        BeforeRequest?.Invoke(p);
        if (p == BlockPath || r.Method + " " + p == BlockPath) { Blocked.TrySetResult(); await Task.Delay(Timeout.Infinite, ct); }
        if (Redirect) return new(HttpStatusCode.Redirect) { Headers = { Location = new Uri("https://example.invalid/") } };
        string repo = "/repositories/" + RepoId;
        if (!p.StartsWith(repo)) throw new Exception("Request outside the selected repository: " + p);
        if (p == repo) return MissingRepo ? new(HttpStatusCode.NotFound) : Json(RepoJson());
        if (p == repo + "/collaborators")
        {
            if (r.RequestUri.Query != "?affiliation=all&per_page=100") throw new Exception("Collaborator query");
            var people = new List<object> { Person(Owner, "participant", true) };
            if (OrganizerAccepted && Owner != GitHubSubmission.ResearcherId) people.Add(Person(OrganizerSpoofId ?? GitHubSubmission.ResearcherId, GitHubSubmission.Researcher, !OrganizerReadOnly));
            if (ExtraCollaborator.HasValue) people.Add(Person(ExtraCollaborator.Value, "someone-else", true));
            var response = Json(people); if (Paginate) response.Headers.Add("Link", "<https://api.github.com/x?page=2>; rel=\"next\""); return response;
        }
        const string prefix = "/contents/";
        if (p.StartsWith(repo + prefix))
        {
            string path = p[(repo.Length + prefix.Length)..];
            if (r.Method == HttpMethod.Put)
            {
                Puts++;
                using var doc = JsonDocument.Parse(body);
                if (doc.RootElement.TryGetProperty("sha", out _) || doc.RootElement.TryGetProperty("branch", out _)) throw new Exception("Overwrite-capable PUT attempted");
                if (PutStatus.HasValue) return new(PutStatus.Value);
                if (Files.ContainsKey(path)) return new(HttpStatusCode.UnprocessableEntity);
                if (FailUploadBeforeStore) { FailUploadBeforeStore = false; throw new HttpRequestException("simulated failure before store"); }
                Files[path] = Convert.FromBase64String(doc.RootElement.GetProperty("content").GetString()!);
                if (FailUploadAfterStore) { FailUploadAfterStore = false; throw new HttpRequestException("simulated lost upload response"); }
                return Json(new { content = new { path, sha = BlobSha(Files[path]) } }, HttpStatusCode.Created);
            }
            if (!r.Headers.Accept.Any(v => v.MediaType == "application/vnd.github.object+json")) throw new Exception("Large submission metadata requires GitHub object media type");
            if (!Files.TryGetValue(path, out var payload)) return new(HttpStatusCode.NotFound);
            return Json(new { type = "file", path, name = Path.GetFileName(path), size = payload.Length, sha = BlobSha(payload), content = "", encoding = "none" });
        }
        if (p.StartsWith(repo + "/git/blobs/"))
        {
            string sha = p[(repo.Length + "/git/blobs/".Length)..];
            var match = Files.Values.FirstOrDefault(v => BlobSha(v) == sha);
            return match == null ? new(HttpStatusCode.NotFound) : new(HttpStatusCode.OK) { Content = new ByteArrayContent(match) };
        }
        throw new Exception("Unexpected fake request: " + r.Method + " " + p);
    }
}
// Route-based synthetic GitHub App sign-in for the real GitHubDeviceAuth.
internal sealed class AuthResponses : HttpMessageHandler
{
    internal int Seconds = 1;
    internal int Revalidations;
    internal long RepositoryId = 987;
    internal string Selection = "selected", Permissions = "{\"contents\":\"write\",\"metadata\":\"read\"}", SuspendedAt = "null";
    protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage r, CancellationToken ct)
    {
        string url = r.RequestUri!.AbsoluteUri.Split('?')[0]; object data;
        if (url.EndsWith("/login/device/code")) data = new { device_code = new string('a', 40), user_code = "ABCD-EFGH", verification_uri = "https://github.com/login/device", expires_in = 900, interval = 5 };
        else if (url.EndsWith("/login/oauth/access_token")) data = new { access_token = FakeGitHub.Token, token_type = "bearer", scope = "", expires_in = Seconds };
        else if (url == "https://api.github.com/user") { Revalidations++; data = new { id = 42, login = "participant", type = "User" }; }
        else if (url == "https://api.github.com/user/installations") data = JsonDocument.Parse("""{"total_count":1,"installations":[{"id":789,"app_id":456,"app_slug":"synthetic-app","target_id":42,"target_type":"User","repository_selection":"SEL","account":{"id":42,"login":"participant","type":"User"},"permissions":PERMS,"suspended_at":SUSP,"suspended_by":null}]}""".Replace("SEL", Selection).Replace("PERMS", Permissions).Replace("SUSP", SuspendedAt)).RootElement;
        else if (url == "https://api.github.com/user/installations/789/repositories") data = JsonDocument.Parse("""{"total_count":1,"repositories":[{"id":REPO,"name":"nose-calibration-submissions","full_name":"participant/nose-calibration-submissions","private":true,"visibility":"private","fork":false,"archived":false,"disabled":false,"owner":{"id":42,"login":"participant","type":"User"},"permissions":{"push":true,"pull":true}}]}""".Replace("REPO", RepositoryId.ToString())).RootElement;
        else throw new Exception("Unexpected auth request " + url);
        return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK) { Content = new StringContent(JsonSerializer.Serialize(data)) });
    }
}

            }
            // Changed local ZIP or envelope after state commit.
            {
                string run = MakeRun(); var fake = new FakeGitHub { FailUploadBeforeStore = true }; using var http = new HttpClient(fake);

            }
            // Crash between state commit and envelope rename: staged envelope is promoted after digest check.
            {
                string run = MakeRun(); var fake = new FakeGitHub { FailUploadBeforeStore = true }; using var http = new HttpClient(fake);

                byte[] frozen = File.ReadAllBytes(Path.Combine(run, "submission.nose"));
                File.Move(Path.Combine(run, "submission.nose"), Path.Combine(run, "submission.nose.tmp"));
                await Upload(run, http); Check(fake.Files[Remote(run)].SequenceEqual(frozen), "staged envelope promoted");
            }
            // Cancellation at each stage propagates and blocks later writes.
            foreach (string block in new[] { "/repositories/987", "/repositories/987/collaborators", "GET /repositories/987/contents/PATH", "PUT /repositories/987/contents/PATH" })
            {
                string run = MakeRun(); var fake = new FakeGitHub { BlockPath = block.Replace("PATH", Remote(run)) }; using var http = new HttpClient(fake); using var cancel = new CancellationTokenSource();
                var upload = Upload(run, http, ct: cancel.Token); await fake.Blocked.Task.WaitAsync(TimeSpan.FromSeconds(5)); cancel.Cancel();
                bool cancelled = false; try { await upload.WaitAsync(TimeSpan.FromSeconds(5)); } catch (OperationCanceledException) { cancelled = true; }
                Check(cancelled && fake.Files.Count == 0 && !State(run).Uploaded, "cancellation at " + block); count++;
            }
            {
                string run = MakeRun(); var fake = new FakeGitHub(); using var http = new HttpClient(fake); using var cancel = new CancellationTokenSource();

                var upload = Upload(run, http, ct: cancel.Token, revalidate: async ct => { entered.SetResult(); await Task.Delay(Timeout.Infinite, ct); return Selected; });

                string run = MakeRun(); byte[]? frozen = null;

            }

        }
