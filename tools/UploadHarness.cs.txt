internal sealed class FakeGitHub : HttpMessageHandler
{
    internal bool Exists, Public, FailCreate, FailUpload, WrongPermission, Redirect;
    internal long Owner = 42;
    internal int Creates, Uploads, Invitations;
    internal HttpStatusCode? CreateStatus;
    internal string? BlockPath;
    internal Action<string>? BeforeRequest;
    internal readonly TaskCompletionSource Blocked = new();
    internal byte[]? Payload;
    readonly string name;
    internal FakeGitHub(string session) { name="nose-calibration-submission-"+session; }
    object Repo() => new {id=987L,name,@private=!Public,owner=new{id=Owner,type="User"},created_at=DateTimeOffset.UtcNow};
    static HttpResponseMessage Json(object data,HttpStatusCode status=HttpStatusCode.OK) => new(status){Content=new StringContent(JsonSerializer.Serialize(data))};
    static HttpResponseMessage Missing() => new(HttpStatusCode.NotFound);
    protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage r,CancellationToken ct)
    {
        string p=r.RequestUri!.AbsolutePath;
        BeforeRequest?.Invoke(p);
        if(p==BlockPath) { Blocked.TrySetResult(); await Task.Delay(Timeout.Infinite,ct); }
        if(Redirect)return new(HttpStatusCode.Redirect){Headers={Location=new Uri("https://example.invalid/")}};
        if(p=="/user")return Json(new{id=Owner,login="participant"});
        if(p.StartsWith("/repos/"))return Exists?Json(Repo()):Missing();
        if(p=="/user/repos") { Creates++;if(CreateStatus.HasValue)return new(CreateStatus.Value);Exists=true;if(FailCreate){FailCreate=false;throw new HttpRequestException("simulated lost create response");}return Json(Repo(),HttpStatusCode.Created); }
        if(p=="/repositories/987")return Json(Repo());
        if(p.EndsWith("/contents/submission.nose"))
        {
            if(r.Method==HttpMethod.Put)
            {
                Uploads++;using var doc=JsonDocument.Parse(await r.Content!.ReadAsStringAsync(ct));
                Payload=Convert.FromBase64String(doc.RootElement.GetProperty("content").GetString()!);
                if(FailUpload){FailUpload=false;throw new HttpRequestException("simulated lost upload response");}
                return Json(new{});
            }
            if(Payload==null)return Missing();
            byte[] prefix=Encoding.ASCII.GetBytes("blob "+Payload.Length+"\0");
            string sha=Convert.ToHexString(SHA1.HashData(prefix.Concat(Payload).ToArray())).ToLowerInvariant();
            return Json(new{type="file",path="submission.nose",size=Payload.Length,sha});
        }
        if(p.Contains("/git/blobs/"))return new(HttpStatusCode.OK){Content=new ByteArrayContent(Payload!)};
        if(p=="/users/"+GitHubSubmission.Researcher)return Json(new{id=GitHubSubmission.ResearcherId});
        if(p.EndsWith("/collaborators/"+GitHubSubmission.Researcher))
        {
            Invitations++;
            using var doc=JsonDocument.Parse(await r.Content!.ReadAsStringAsync(ct));
            if(doc.RootElement.GetProperty("permission").GetString()!="push")throw new Exception("Must disclose and request write");
            return Json(new{invitee=new{id=GitHubSubmission.ResearcherId},permissions=WrongPermission?"read":"write"},HttpStatusCode.Created);
        }
        throw new Exception("Unexpected fake request: "+p);
    }
}
internal sealed class AuthResponses : HttpMessageHandler
{
    int request;
    internal int Seconds = 1;
    protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage r,CancellationToken ct)
    {
        object data = ++request switch
        {
            1 => new {device_code=new string('a',40),user_code="ABCD-EFGH",verification_uri="https://github.com/login/device",expires_in=900,interval=5},
            2 => new {access_token="gho_syntheticTESTtoken00000000000000000000",token_type="bearer",scope="repo",expires_in=Seconds},
            _ => new {id=42,login="participant",type="User"}
        };
        return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK){Content=new StringContent(JsonSerializer.Serialize(data))});
    }
}
